        function updateTimeSlotAvailability() {
            const dateVal = document.getElementById('bDate').value;
            const timeSelect = document.getElementById('bTime');
            const infoEl = document.getElementById('timeSlotInfo');
            if (!timeSelect) return;

            const selectedTimeVal = timeSelect.value;

            if (!dateVal) {
                timeSelect.innerHTML = '<option value="">Önce Tarih Seçiniz</option>';
                ALL_TIME_SLOTS.forEach(time => {
                    const opt = document.createElement('option');
                    opt.value = time;
                    opt.textContent = time;
                    timeSelect.appendChild(opt);
                });
                if (infoEl) infoEl.innerText = '';
                return;
            }

            const bookedTimes = apps
                .filter(a => a.date === dateVal && a.status !== 'İptal')
                .map(a => a.time);

            let availableCount = 0;
            timeSelect.innerHTML = '<option value="">Saat Seçiniz</option>';

            ALL_TIME_SLOTS.forEach(time => {
                const isBooked = bookedTimes.includes(time);
                const option = document.createElement('option');
                option.value = time;

                if (isBooked) {
                    option.textContent = `${time} - (DOLU / REZERVE)`;
                    option.disabled = true;
                    option.className = 'text-gray-600 bg-gray-900 font-normal';
                } else {
                    option.textContent = `${time} (Müsait)`;
                    availableCount++;
                }

                if (time === selectedTimeVal && !isBooked) {
                    option.selected = true;
                }

                timeSelect.appendChild(option);
            });

            if (infoEl) {
                if (availableCount === 0) {
                    infoEl.innerText = '⚠️ Bu tarihteki tüm randevu saatlerimiz doludur.';
                    infoEl.className = 'text-xs text-red-400 mt-1.5 block font-medium';

            }

        function formatDuration(mins) {
            if (mins <= 0) return '0 Dk';
            const hours = Math.floor(mins / 60);
            const remainingMins = mins % 60;
            if (hours > 0 && remainingMins > 0) {
                return `${hours} Saat ${remainingMins} Dk`;
            } else if (hours > 0) {
                return `${hours} Saat`;
            }
            return `${remainingMins} Dk`;
        }

            tomorrow.setDate(tomorrow.getDate() + 1);
            const tomorrowStr = tomorrow.toISOString().split('T')[0];

            const myApproved = apps.filter(a => a.name && currentUser && a.name.toLowerCase() === currentUser.toLowerCase() && a.status === 'Onaylandı');
            const banner = document.getElementById('reminderBanner');
            const reminderText = document.getElementById('reminderText');

            let reminderMsg = '';
            myApproved.forEach(app => {
                if (app.date === todayStr) {
                    reminderMsg = `Hatırlatma: Bugün saat ${app.time}'de "${app.service}" randevunuz bulunmaktadır! Bekliyoruz.`;
                } else if (app.date === tomorrowStr) {
                    reminderMsg = `Hatırlatma: Yarın saat ${app.time}'de "${app.service}" randevunuz bulunmaktadır!`;
                }
            });

            if (reminderMsg) {
                if (reminderText) reminderText.innerText = reminderMsg;
                if (banner) banner.classList.remove('hidden');
            } else {
                if (banner) banner.classList.add('hidden');
            }
        }

        function handleBooking(e) {
            e.preventDefault();
            const warningEl = document.getElementById('bookingWarning');
            const successEl = document.getElementById('bookingSuccess');
            if (warningEl) warningEl.classList.add('hidden');
            if (successEl) successEl.classList.add('hidden');

            if (!checkUserLogin()) {
                if (warningEl) warningEl.classList.remove('hidden');
                openLoginModal();
                return;
            }

            if (selectedServiceIds.length === 0) {
                alert('Lütfen en az bir hizmet seçiniz!');
                return;
            }

            const selectedDate = document.getElementById('bDate').value;
            const selectedTime = document.getElementById('bTime').value;

            if (!selectedDate || !selectedTime) {
                alert('Lütfen geçerli bir tarih ve saat seçiniz.');
                return;
            }

            // Seçilen saat dolu mu kontrol et
            const currentApps = getApps();
            const isSlotTaken = currentApps.some(a => a.date === selectedDate && a.time === selectedTime && a.status !== 'İptal');
            if (isSlotTaken) {
                alert('Seçtiğiniz tarih ve saat dilimi doludur. Lütfen farklı bir saat seçiniz.');
                updateTimeSlotAvailability();
                return;
            }

            let totalMin = 0;
            let totalPrice = 0;
            let selectedObjects = [];

            selectedServiceIds.forEach(id => {
                const srv = SERVICES_DATA.find(s => s.id === id);
                if (srv) {
                    totalMin += srv.durationMin;
                    totalPrice += srv.price;
                    selectedObjects.push(srv);
                }
            });

            const serviceNames = selectedObjects.map(s => s.name).join(' + ');
            const fullServiceSummary = `${serviceNames} (${formatDuration(totalMin)} - ${totalPrice.toLocaleString('tr-TR')} ₺)`;

            const fileInput = document.getElementById('bFile');

            const saveRecord = (img64) => {
                const apps = getApps();
                const newRecord = {
                    id: Date.now(),
                    name: currentUser,
                    phone: document.getElementById('bPhone').value,
                    service: fullServiceSummary,
                    serviceList: selectedObjects,
                    totalPrice: totalPrice,
                    totalDurationMin: totalMin,
                    date: selectedDate,
                    time: selectedTime,
                    note: document.getElementById('bNote').value,
                    image: img64,
                    status: 'Beklemede',
                    reply: ''
                };
                apps.push(newRecord);
                saveApps(apps);
                e.target.reset();
                selectedServiceIds = [];
                renderServiceSelectionGrid();
                updateTimeSlotAvailability();
                
                if (successEl) {
                    successEl.classList.remove('hidden');
                    setTimeout(() => successEl.classList.add('hidden'), 4000);
                }
                renderUserAppointments();
            };

            if (fileInput.files && fileInput.files[0]) {
                const reader = new FileReader();
                reader.onload = function(uploadEvent) {
                    saveRecord(uploadEvent.target.result);
                };
                reader.readAsDataURL(fileInput.files[0]);
            } else {
                saveRecord('');
            }
        }

        function renderAdminUsersTable() {
            const users = getUsersList();
            const apps = getApps();
            const tbody = document.getElementById('adminUsersTableBody');
            if (!tbody) return;
            tbody.innerHTML = '';

            if (users.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" class="text-center p-6 text-gray-500">Henüz kayıtlı kullanıcı bulunmuyor.</td></tr>';
                return;
            }

            users.forEach((u) => {
                const userApps = apps.filter(a => a.name && a.name.toLowerCase() === u.name.toLowerCase());
                const userAppCount = userApps.length;
                
                let totalSpent = 0;
                userApps.forEach(app => {
                    if (app.status === 'Onaylandı') {
                        if (app.totalPrice) {
                            totalSpent += app.totalPrice;
                        } else if (app.service) {
                            const match = app.service.match(/([\d\.]+)\s*₺/);
                            if (match) {
                                const val = parseInt(match[1].replace(/\./g, ''), 10);
                                if (!isNaN(val)) totalSpent += val;
                            }
                        }
                    }
                });

                const formattedSpent = new Intl.NumberFormat('tr-TR').format(totalSpent);

                const tr = document.createElement('tr');
                tr.className = "hover:bg-gray-900/50";
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${u.name}</td>
                    <td class="p-3 text-gray-400">${u.pass}</td>
                    <td class="p-3"><span class="px-2.5 py-1 rounded-full text-xs bg-[#C5A059]/10 text-[#C5A059]">${userAppCount} Adet</span></td>
                    <td class="p-3"><span class="px-2.5 py-1 rounded-full text-xs bg-emerald-900/40 text-emerald-300 border border-emerald-800 font-semibold">${formattedSpent} ₺</span></td>
                    <td class="p-3">
                        <button onclick="deleteUserAccount('${u.name}')" class="text-red-400 hover:text-red-300 text-xs bg-red-950/40 border border-red-900 px-3 py-1.5 rounded-lg flex items-center gap-1">
                            <i class="fa-solid fa-trash"></i> Hesabı Sil
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function deleteUserAccount(userName) {
            if (confirm(`"${userName}" adlı kullanıcı hesabını ve ilişkili tüm randevuları silmek istediğinize emin misiniz?`)) {
                let users = getUsersList();
                users = users.filter(u => u.name.toLowerCase() !== userName.toLowerCase());
                saveUsersList(users);

                let apps = getApps();
                apps = apps.filter(a => a.name.toLowerCase() !== userName.toLowerCase());
                saveApps(apps);

                if (currentUser && currentUser.toLowerCase() === userName.toLowerCase()) {
                    handleLogout();
                }

                renderAdminUsersTable();
                renderAdminTable();
            }
        }

        function handleRegisterSubmit() {
            const name = document.getElementById('regName').value.trim();
            const pass = document.getElementById('regPass').value;
            const err = document.getElementById('regErrorMsg');

            if(!name || pass.length < 6 || !/[A-Z]/.test(pass)) {
                err.innerText = "Şifre en az 6 karakter olmalı ve 1 büyük harf içermelidir!";
                err.classList.remove('hidden');
                return;
            }

            let users = getUsersList();
            if (users.some(u => u.name.toLowerCase() === name.toLowerCase())) {
                err.innerText = "Bu isimde bir kullanıcı zaten mevcut!";
                err.classList.remove('hidden');
                return;
            }

            users.push({ name, pass });
            saveUsersList(users);

            localStorage.setItem('nes_reg_name', name);
            localStorage.setItem('nes_reg_pass', pass);

            closeRegisterModal();
            openLoginModal();
        }

        function handleLoginSubmit() {
            const name = document.getElementById('loginName').value.trim();
            const pass = document.getElementById('loginPass').value;
            const err = document.getElementById('loginErrorMsg');
            
            let users = getUsersList();
            const foundUser = users.find(u => u.name.toLowerCase() === name.toLowerCase() && u.pass === pass);

            // Geriye dönük uyumluluk için eski tekli kayıt kontrolü
            const savedName = localStorage.getItem('nes_reg_name');
            const savedPass = localStorage.getItem('nes_reg_pass');
            const isOldMatch = (savedName && name === savedName && pass === savedPass);

            if(foundUser || isOldMatch) {
                currentUser = name;
                localStorage.setItem('nes_logged_user', name);
                
                if (!users.some(u => u.name.toLowerCase() === name.toLowerCase())) {
                    users.push({ name, pass });
                    saveUsersList(users);
                }

                closeLoginModal();
                updateNavAuthUI();
            } else {
                err.innerText = "Hatalı Ad Soyad veya Şifre!";
                err.classList.remove('hidden');
            }
        }

        function handleLogout() {
            currentUser = null;
            localStorage.removeItem('nes_logged_user');
            updateNavAuthUI();
        }
